# Исправленное доказательство из методички. Аксиомы известны заранее.
/goal p -> p
# 1. A2, p := r
(s -> (r -> q)) -> ((s -> r) -> (s -> q))
# 2. Шаг 1, s := p
(p -> (r -> q)) -> ((p -> r) -> (p -> q))
# 3. Шаг 2, q := p
(p -> (r -> p)) -> ((p -> r) -> (p -> p))
# 4. A1, q := r
p -> (r -> p)
# 5. MP: шаги 3 и 4. Скобки вокруг (p -> r) обязательны.
(p -> r) -> (p -> p)
# 6. Шаг 5, r := (q -> p)
(p -> (q -> p)) -> (p -> p)
# 7. MP: шаг 6 и A1
p -> p
